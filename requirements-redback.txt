# Environment for src/fitting/redback-fit and src/ingest.
#
# Versions are pinned so a re-run of `make setup-redback` reproduces the fits
# published from this repo. Bump them deliberately, then re-run
# redback_models.check_against_redback() -- it reports any prior key a redback
# upgrade renamed, instead of letting it silently change a fit.
#
# dustmaps is absent on purpose: src/ingest/extinction.py is imported only by
# src/fitting/nmma-fit, so it lives in requirements-nmma.txt.

# The afterglow extra pulls afterglowpy; redback itself brings bilby, sncosmo
# and extinction.
redback[afterglow]==1.20.0

# Samplers: pymultinest is the default, dynesty the pure-Python fallback
# (fit_nickel.py --sampler). pymultinest loads libmultinest at import time,
# so `make setup-multinest` must have run first.
pymultinest==2.12
dynesty==3.1.0
bilby==2.8.2

# Imported directly by redback-fit (plot_redback, fit_nickel) and by
# src/ingest (fetch_ztf, time_windows); transitive through redback too, but
# pinned here because this code calls them itself.
numpy==2.5.4
pandas==3.0.6
astropy==8.0.1
matplotlib==3.11.2
